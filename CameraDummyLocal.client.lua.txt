--============================================================================
-- CameraDummyLocal.client.lua  (LOCALSCRIPT — 100% CLIENTE)
--============================================================================
-- Sistema "Camera Dummy Voadora" — TOTALMENTE LOCAL (sem script de servidor).
--
-- Tudo é feito pelo cliente:
--   • Cria o Dummy direto no Workspace.
--   • Aplica o acessório de câmera (Asset 116029058192063).
--   • Esconde o corpo do Dummy (só a câmera fica visível).
--   • Controla o FLY (WASD + vertical + direção da câmera).
--   • Alterna a câmera entre Player e Dummy (primeira pessoa).
--   • Limpa tudo ao morrer/respawnar/fechar.
--
-- O jogador REAL nunca é movido. O Dummy atravessa paredes (sem colisão).
-- Usa apenas APIs oficiais do Roblox.
--============================================================================

--==================== SERVIÇOS ====================
local Players = game:GetService("Players")
local player  = Players.LocalPlayer
local UIS     = game:GetService("UserInputService")
local RunService = game:GetService("RunService")
local Workspace = game:GetService("Workspace")
local InsertService = game:GetService("InsertService")
local camera = Workspace.CurrentCamera

--==================== CONFIGURAÇÃO ====================
local CAMERA_ACCESSORY_ID = 116029058192063 -- acessório de câmera

local DEFAULT_SPEED = 50     -- velocidade padrão (studs/s)
local MIN_SPEED     = 1
local MAX_SPEED     = 500

local MOUSE_SENSITIVITY = 0.0025 -- radianos por pixel (mouse-look)

--==================== ESTADO ====================
local dummy = nil
local flyEnabled = false
local dummyView = false
local flySpeed = DEFAULT_SPEED

local yaw = 0
local pitch = 0

local upHeld = false   -- botão UP pressionado
local downHeld = false -- botão DOWN pressionado

-- CFrame "mestre" do Dummy (posição/orientação atuais)
local masterCF = CFrame.new()

-- Lista de partes que movemos juntas: { part, offset }
-- offset = posição local da parte em relação ao CFrame mestre
local trackedParts = {}

-- Lista de conexões (para desconectar tudo ao desativar)
local connections = {}
local function connect(sig, fn)
	local c = sig:Connect(fn)
	table.insert(connections, c)
	return c
end

--==================== REFERÊNCIAS DA GUI ====================
local screenGui, flyButton, viewButton, speedBox

local function updateFlyButton()
	if flyButton then
		flyButton.Text = flyEnabled and "Fly: ON" or "Fly: OFF"
	end
end

local function updateViewButton()
	if viewButton then
		viewButton.Text = dummyView and "Visão: Dummy" or "Visão: Player"
	end
end

--==================== CRIAÇÃO DO DUMMY ====================

-- Posiciona uma parte como parte do "rig" invisível do Dummy
local function trackPart(part, offset)
	part.Anchored   = true   -- drone: sem gravidade, controlado por CFrame
	part.CanCollide = false  -- sem colisão (atravessa paredes)
	part.CanQuery   = false  -- não é atingido por raycasts
	part.CanTouch   = false  -- não dispara eventos de toque
	part.Transparency = 1    -- corpo invisível
	part.CFrame = masterCF * offset
	part.Parent = dummy
	table.insert(trackedParts, { part = part, offset = offset })
end

local function newBodyPart(name, size, offset)
	local p = Instance.new("Part")
	p.Name = name
	p.Size = size
	trackPart(p, offset)
	return p
end

-- Move TODAS as partes do Dummy para um novo CFrame mestre
local function setDummyCFrame(cf)
	masterCF = cf
	for _, entry in ipairs(trackedParts) do
		if entry.part and entry.part.Parent then
			entry.part.CFrame = cf * entry.offset
		end
	end
end

-- Aplica o acessório de câmera (assíncrono)
local function applyAccessory()
	local ok, result = pcall(function()
		return InsertService:LoadAsset(CAMERA_ACCESSORY_ID)
	end)

	if ok and result then
		-- Procura o Handle (a malha da câmera)
		local handle = result:FindFirstChild("Handle", true)
		if not handle then
			handle = result:FindFirstChildWhichIsA("BasePart")
		end

		if handle then
			-- Remove scripts internos (evita comportamento indesejado)
			for _, c in ipairs(handle:GetChildren()) do
				if c:IsA("Script") or c:IsA("LocalScript") then
					c:Destroy()
				end
			end

			handle.Parent = dummy
			-- Offset da câmera: fica na altura da "cabeça" do Dummy
			trackPart(handle, CFrame.new(0, 1.6, 0))
		end
		result:Destroy() -- descarta o resto do acessório
	else
		warn("[CameraDummy] Falha ao carregar o acessório:", result)
		makeFallbackCamera()
	end
end

-- Fallback: se o acessório falhar, cria uma "câmera" simples
local function makeFallbackCamera()
	local cam = Instance.new("Part")
	cam.Name = "CameraFallback"
	cam.Size = Vector3.new(0.8, 0.6, 0.8)
	cam.BrickColor = BrickColor.new("Dark stone grey")
	trackPart(cam, CFrame.new(0, 1.6, 0))
end

local function createDummy()
	-- Remove qualquer Dummy antigo primeiro
	destroyDummy()

	local character = player.Character
	if not character then return nil end

	-- Posição inicial: onde o jogador está
	local spawnCF = CFrame.new(0, 10, 0)
	local hrp = character:FindFirstChild("HumanoidRootPart")
	if hrp then
		spawnCF = hrp.CFrame
	else
		spawnCF = character:GetPivot()
	end
	masterCF = spawnCF

	dummy = Instance.new("Model")
	dummy.Name = "CameraDummy_" .. player.Name
	dummy.Parent = Workspace

	-- Rig invisível (só a câmera fica visível)
	newBodyPart("HumanoidRootPart", Vector3.new(2, 2, 1), CFrame.new())
	newBodyPart("Torso", Vector3.new(2, 2, 1), CFrame.new())
	newBodyPart("Head", Vector3.new(1.5, 1.2, 1), CFrame.new(0, 1.6, 0))

	-- Aplica o acessório de câmera
	applyAccessory()

	return dummy
end

local function destroyDummy()
	for _, entry in ipairs(trackedParts) do
		if entry.part and entry.part.Parent then
			entry.part:Destroy()
		end
	end
	trackedParts = {}
	if dummy then
		dummy:Destroy()
	end
	dummy = nil
end

--==================== CÂMERA ====================

-- Posição da "lente" (Handle do acessório, ou cabeça, ou root)
local function getCameraAnchor()
	if not dummy or not dummy.Parent then return nil end
	for _, entry in ipairs(trackedParts) do
		local p = entry.part
		if p and p.Parent and p.Name == "Handle" then
			return p.Position
		end
	end
	for _, entry in ipairs(trackedParts) do
		local p = entry.part
		if p and p.Parent and p.Name == "Head" then
			return p.Position
		end
	end
	return masterCF.Position
end

-- Atualiza a câmera do Dummy (primeira pessoa) continuamente
local function updateDummyCamera()
	local anchorPos = getCameraAnchor()
	if not anchorPos then return end

	camera.CameraType = Enum.CameraType.Scriptable
	camera.CFrame = CFrame.new(anchorPos) * CFrame.Angles(0, yaw, 0) * CFrame.Angles(pitch, 0, 0)

	-- Orienta o Dummy para "olhar" na mesma direção da câmera
	setDummyCFrame(CFrame.new(masterCF.Position) * CFrame.Angles(0, yaw, 0) * CFrame.Angles(pitch, 0, 0))
end

local function restorePlayerCamera()
	camera.CameraType = Enum.CameraType.Custom
	local char = player.Character
	if char then
		local hum = char:FindFirstChildOfClass("Humanoid")
		if hum then
			camera.CameraSubject = hum
		end
	end
	UIS.MouseBehavior = Enum.MouseBehavior.Default
end

--==================== MOVIMENTO / FLY ====================

local function getMoveVector()
	local fwd = camera.CFrame.LookVector
	local right = camera.CFrame.RightVector

	-- Achata no plano horizontal (vertical tratado separado)
	local f = Vector3.new(fwd.X, 0, fwd.Z)
	local r = Vector3.new(right.X, 0, right.Z)
	if f.Magnitude < 0.001 then f = Vector3.new(0, 0, -1) end
	if r.Magnitude < 0.001 then r = Vector3.new(1, 0, 0) end
	f = f.Unit
	r = r.Unit

	local move = Vector3.new(0, 0, 0)
	if UIS:IsKeyDown(Enum.KeyCode.W) then move = move + f end
	if UIS:IsKeyDown(Enum.KeyCode.S) then move = move - f end
	if UIS:IsKeyDown(Enum.KeyCode.A) then move = move - r end
	if UIS:IsKeyDown(Enum.KeyCode.D) then move = move + r end
	if UIS:IsKeyDown(Enum.KeyCode.Space) or UIS:IsKeyDown(Enum.KeyCode.E) then
		move = move + Vector3.new(0, 1, 0)
	end
	if UIS:IsKeyDown(Enum.KeyCode.LeftShift) or UIS:IsKeyDown(Enum.KeyCode.Q) then
		move = move - Vector3.new(0, 1, 0)
	end
	if upHeld then move = move + Vector3.new(0, 1, 0) end
	if downHeld then move = move - Vector3.new(0, 1, 0) end

	return move
end

local function moveDummy(dt)
	local move = getMoveVector()
	if move.Magnitude < 0.001 then return end

	local translation = move.Unit * flySpeed * dt

	if dummyView then
		-- Em visão dummy a rotação já foi definida; só translada
		setDummyCFrame(masterCF + translation)
	else
		-- Fora da visão, orienta o drone na direção horizontal do movimento
		local horiz = Vector3.new(translation.X, 0, translation.Z)
		if horiz.Magnitude > 0.001 then
			setDummyCFrame(CFrame.lookAt(
				masterCF.Position + translation,
				masterCF.Position + translation + horiz.Unit
			))
		else
			setDummyCFrame(masterCF + translation)
		end
	end
end

--==================== VISÃO (Player x Dummy) ====================

local function enterDummyView()
	if not dummy or not dummy.Parent then return end
	dummyView = true

	local rx, ry = camera.CFrame:ToEulerAnglesYXZ()
	yaw = ry
	pitch = rx

	camera.CameraType = Enum.CameraType.Scriptable
	UIS.MouseBehavior = Enum.MouseBehavior.LockCenter
end

local function exitDummyView()
	dummyView = false
	restorePlayerCamera()
end

--==================== LIMPEZA ====================

local function cleanup()
	flyEnabled = false
	dummyView = false
	destroyDummy()
	restorePlayerCamera()
	updateFlyButton()
	updateViewButton()
end

local function disable()
	cleanup()
	for _, c in ipairs(connections) do
		c:Disconnect()
	end
	connections = {}
	if screenGui then
		screenGui:Destroy()
	end
end

--==================== GUI ====================

local function buildGui()
	screenGui = Instance.new("ScreenGui")
	screenGui.Name = "CameraDummyGUI"
	screenGui.ResetOnSpawn = false
	screenGui.ZIndexBehavior = Enum.ZIndexBehavior.Sibling
	screenGui.Parent = player:WaitForChild("PlayerGui")

	local frame = Instance.new("Frame")
	frame.Name = "Main"
	frame.Size = UDim2.new(0, 220, 0, 250)
	frame.Position = UDim2.new(0.01, 0, 0.25, 0)
	frame.BackgroundColor3 = Color3.fromRGB(28, 28, 32)
	frame.BorderSizePixel = 0
	frame.Active = true
	frame.Draggable = true
	frame.Parent = screenGui

	local function label(text, y, size, color, bold)
		local l = Instance.new("TextLabel")
		l.BackgroundTransparency = 1
		l.Size = UDim2.new(1, -10, 0, size)
		l.Position = UDim2.new(0, 5, 0, y)
		l.Font = bold and Enum.Font.SourceSansBold or Enum.Font.SourceSans
		l.Text = text
		l.TextColor3 = color or Color3.fromRGB(255, 255, 255)
		l.TextSize = size - 2
		l.TextXAlignment = Enum.TextXAlignment.Left
		l.Parent = frame
		return l
	end

	local function button(text, y, color, onClick)
		local b = Instance.new("TextButton")
		b.Size = UDim2.new(1, -10, 0, 28)
		b.Position = UDim2.new(0, 5, 0, y)
		b.BackgroundColor3 = color or Color3.fromRGB(60, 60, 70)
		b.BorderSizePixel = 0
		b.Font = Enum.Font.SourceSansBold
		b.Text = text
		b.TextColor3 = Color3.fromRGB(0, 0, 0)
		b.TextSize = 14
		b.Parent = frame
		if onClick then
			connect(b.MouseButton1Click, onClick)
		end
		return b
	end

	label("CAMERA DUMMY (LOCAL)", 6, 18, Color3.fromRGB(163, 255, 137), true)

	-- 1) Spawnar Câmera
	button("Spawnar Câmera", 30, Color3.fromRGB(79, 255, 152), function()
		dummy = createDummy()
		updateFlyButton()
		updateViewButton()
	end)

	-- 2) Regerar Dummy
	button("Regerar Dummy", 62, Color3.fromRGB(120, 220, 255), function()
		dummy = createDummy()
		if dummyView and dummy then
			camera.CameraType = Enum.CameraType.Scriptable
			UIS.MouseBehavior = Enum.MouseBehavior.LockCenter
		end
		updateFlyButton()
		updateViewButton()
	end)

	-- 3) Toggle Fly
	flyButton = button("Fly: OFF", 94, Color3.fromRGB(255, 249, 74), function()
		flyEnabled = not flyEnabled
		updateFlyButton()
	end)

	-- 4) Velocidade (TextBox)
	label("Velocidade:", 126, 16)
	speedBox = Instance.new("TextBox")
	speedBox.Size = UDim2.new(0, 60, 0, 26)
	speedBox.Position = UDim2.new(0, 110, 0, 126)
	speedBox.BackgroundColor3 = Color3.fromRGB(45, 45, 52)
	speedBox.BorderSizePixel = 0
	speedBox.TextColor3 = Color3.fromRGB(255, 255, 255)
	speedBox.Font = Enum.Font.SourceSans
	speedBox.TextSize = 16
	speedBox.Text = tostring(DEFAULT_SPEED)
	speedBox.PlaceholderText = "studs/s"
	speedBox.Parent = frame

	local function applySpeed()
		local n = tonumber(speedBox.Text)
		if n and n > 0 then
			flySpeed = math.clamp(n, MIN_SPEED, MAX_SPEED)
			speedBox.Text = tostring(flySpeed)
		else
			flySpeed = DEFAULT_SPEED
			speedBox.Text = tostring(DEFAULT_SPEED)
		end
	end
	connect(speedBox.FocusLost, applySpeed)
	connect(speedBox.ReturnPressed, function()
		applySpeed()
		speedBox:ReleaseFocus()
	end)

	-- 5) Toggle Visão
	viewButton = button("Visão: Player", 158, Color3.fromRGB(255, 160, 80), function()
		if dummyView then
			exitDummyView()
		else
			if not dummy or not dummy.Parent then
				dummy = createDummy()
				if not dummy then return end
			end
			enterDummyView()
		end
		updateViewButton()
	end)

	-- UP / DOWN (movimento vertical)
	local upBtn = button("UP", 190, Color3.fromRGB(79, 255, 152))
	upBtn.Size = UDim2.new(0, 100, 0, 28)
	upBtn.Position = UDim2.new(0, 5, 0, 190)
	connect(upBtn.MouseButton1Down, function() upHeld = true end)
	connect(upBtn.MouseButton1Up, function() upHeld = false end)
	connect(upBtn.MouseLeave, function() upHeld = false end)

	local downBtn = button("DOWN", 190, Color3.fromRGB(255, 150, 150))
	downBtn.Size = UDim2.new(0, 100, 0, 28)
	downBtn.Position = UDim2.new(0, 115, 0, 190)
	connect(downBtn.MouseButton1Down, function() downHeld = true end)
	connect(downBtn.MouseButton1Up, function() downHeld = false end)
	connect(downBtn.MouseLeave, function() downHeld = false end)

	-- Fechar (X)
	local close = Instance.new("TextButton")
	close.Size = UDim2.new(0, 26, 0, 26)
	close.Position = UDim2.new(1, -30, 0, 4)
	close.BackgroundColor3 = Color3.fromRGB(225, 25, 0)
	close.BorderSizePixel = 0
	close.Font = Enum.Font.SourceSansBold
	close.Text = "X"
	close.TextColor3 = Color3.fromRGB(255, 255, 255)
	close.TextSize = 16
	close.Parent = frame
	connect(close.MouseButton1Click, disable)
end

--==================== ENTRADA (mouse-look) ====================
connect(UIS.InputChanged, function(input, gameProcessed)
	if not dummyView then return end
	if input.UserInputType == Enum.UserInputType.MouseMovement then
		local delta = input.Delta
		yaw = yaw - delta.X * MOUSE_SENSITIVITY
		pitch = math.clamp(pitch - delta.Y * MOUSE_SENSITIVITY, -math.rad(89), math.rad(89))
	end
end)

--==================== LOOP PRINCIPAL ====================
connect(RunService.RenderStepped, function(dt)
	if dummyView then
		updateDummyCamera()
	end
	if flyEnabled and dummy and dummy.Parent then
		moveDummy(dt)
	end
end)

--==================== MORTE / RESPAWN ====================
local function onCharacterAdded(char)
	cleanup()
	local hum = char:WaitForChild("Humanoid", 10)
	if hum then
		connect(hum.Died, function()
			cleanup()
		end)
	end
end

if player.Character then
	task.spawn(onCharacterAdded, player.Character)
end
connect(player.CharacterAdded, onCharacterAdded)

--==================== INICIALIZAÇÃO ====================
buildGui()
